import pandas as pd

df = pd.read_csv("data.csv")

allowed_categories = ["Electronics", "Clothing", "Food", "Furniture"]

def validate_row(row):
    errors = []

    try:
        quantity = int(row["quantity"])
        if quantity < 0:
            errors.append("Quantity cannot be negative")
    except (ValueError, TypeError):
        errors.append("Quantity must be an integer")

    try:
        price = float(row["price"])
        if price <= 0:
            errors.append("Price must be positive")
    except (ValueError, TypeError):
        errors.append("Price must be a number")

    category = str(row["category"]).strip()
    if category not in allowed_categories:
        errors.append("Invalid category")

    if len(errors) == 0:
        return pd.Series({"Flag": "VALID", "Reason": "—"})
    else:
        return pd.Series({"Flag": "INVALID", "Reason": "; ".join(errors)})

result = df.apply(validate_row, axis=1)

df["Flag"] = result["Flag"]
df["Reason"] = result["Reason"]

df.to_csv("validated_data.csv", index=False)
print("Done")
