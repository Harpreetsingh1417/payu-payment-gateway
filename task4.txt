const fileInput = document.getElementById("csvFile");
const uploadBtn = document.getElementById("uploadBtn");
const message = document.getElementById("message");
const tableContainer = document.getElementById("tableContainer");

uploadBtn.onclick = () => {
    const file = fileInput.files[0];

    if (!file) return showError("Please select a file first.");
    if (!file.name.toLowerCase().endsWith(".csv"))
        return showError("Please upload a CSV file.");

    const reader = new FileReader();

    reader.onload = e => {
        const rows = parseCSV(e.target.result);

        if (!rows.length) return showError("The CSV file is empty.");

        message.textContent = "CSV uploaded successfully!";
        message.className = "success";

        renderTable(rows.slice(0, 20));
    };

    reader.onerror = () => showError("Unable to read the file.");
    reader.readAsText(file);
};


// Manual CSV parser
 function parseCSV(text) {
    const rows = [];

    const lines = text.split("\n");

    for (let line of lines) {
        line = line.trim();

        if (line === "") continue;

        const row = line.split(",");
        rows.push(row);
    }

    return rows;
}

// Create table
function renderTable(rows) {
    tableContainer.innerHTML = `
        <table>
            <thead>
                <tr>${rows[0].map(h => `<th>${h}</th>`).join("")}</tr>
            </thead>
            <tbody>
                ${rows.slice(1).map(row =>
                    `<tr>${row.map(cell => `<td>${cell}</td>`).join("")}</tr>`
                ).join("")}
            </tbody>
        </table>`;
}
