function onFormSubmit(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let logSheet = ss.getSheetByName('Log');
  
  if (!logSheet) {
    logSheet = ss.insertSheet('Log');
    logSheet.appendRow(['Serial No.', 'Timestamp', 'Name', 'Country', 'State', 'Status']);
  }
  
  const responses = e.values || []; 
  const name = responses[1] ? responses[1].trim() : '';
  const country = responses[2] ? responses[2].trim() : '';
  const state = responses[3] ? responses[3].trim() : '';

  // Basic Validation: check if any required field is empty
  const isValid = name !== '' && country !== '' && state !== '';
  const status = isValid ? 'Valid' : 'FLAGGED: Missing Data';

  const nextSerialNo = logSheet.getLastRow(); 
  const logTimestamp = new Date();
  
  logSheet.appendRow([
    nextSerialNo,
    logTimestamp,
    name,
    country,
    state,
    status
  ]);
}
