function onFormSubmit(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let logSheet = ss.getSheetByName('Log');
  
  if (!logSheet) {
    logSheet = ss.insertSheet('Log');
    logSheet.appendRow(['Serial No.', 'Timestamp', 'Name', 'Country', 'State', 'Status']);
  }
  
  const responses = e.values || []; 
  const name = responses[1] ? responses[1].trim() : '';
  const country = responses[2] ? responses[2].trim() : '';
  const state = responses[3] ? responses[3].trim() : '';

  // Basic Validation: check if any required field is empty
  const isValid = name !== '' && country !== '' && state !== '';
  const status = isValid ? 'Valid' : 'FLAGGED: Missing Data';

  const nextSerialNo = logSheet.getLastRow(); 
  const logTimestamp = new Date();
  
  logSheet.appendRow([
    nextSerialNo,
    logTimestamp,
    name,
    country,
    state,
    status
  ]);
}

const dropArea = document.getElementById("dropArea");
const fileInput = document.getElementById("csvFile");
const message = document.getElementById("message");
const tableContainer = document.getElementById("tableContainer");

fileInput.addEventListener("change", function () {
    const file = fileInput.files[0];

    if (file) {
        processFile(file);
    }
});

dropArea.addEventListener("click", function () {
    fileInput.click();
});

dropArea.addEventListener("dragover", function (event) {
    event.preventDefault();
    dropArea.classList.add("dragover");
});

dropArea.addEventListener("dragleave", function () {
    dropArea.classList.remove("dragover");
});

dropArea.addEventListener("drop", function (event) {
    event.preventDefault();

    dropArea.classList.remove("dragover");

    const file = event.dataTransfer.files[0];

     if (file) {
         processFile(file);
     }
});

 function processFile(file) {

    if (!file.name.toLowerCase().endsWith(".csv")) {
        showError("Please select a valid CSV file.");
        return;
    }

const reader = new FileReader();

    reader.onload = e => {
        const rows = parseCSV(e.target.result);

        if (!rows.length) return showError("The CSV file is empty.");

        message.textContent = "CSV uploaded successfully!";
        message.className = "success";

        renderTable(rows.slice(0, 21));
};
    reader.onerror = () => showError("Unable to read the file.");
    reader.readAsText(file);
};

function parseCSV(text) {
    const rows = [];
    const lines = text.split("\n");

    for (let line of lines) {
        line = line.trim();

        if (line === "") continue;

        const row = line.split(",");
        rows.push(row);
    }
    return rows;
}

function renderTable(rows) { 
    tableContainer.innerHTML = `
    <table> 
        <thead> <tr>${rows[0].map(h => `<th>${h}</th>`).join("")}</tr> 
        </thead> 
        <tbody> ${rows.slice(1).map(row => `<tr>${row.map(cell =>`<td>${cell}</td>`).join("")}</tr>`).join("")} 
        </tbody> 
        </table>`; 
};
