function onFormSubmit(e) {
  const ss = SpreadsheetApp.getActive();
  const sheet = ss.getSheetByName("Log") || ss.insertSheet("Log");
  const data = e.values.slice(1);

  if (sheet.getLastRow() === 0)
    sheet.appendRow(["Sr. No.", "Timestamp", ...e.range.getSheet().getRange(1, 2, 1, data.length).getValues()[0]]);

  const missing = data.some(x => !x.trim()) ? "Missing" : "";

  sheet.appendRow([sheet.getLastRow(), new Date(), ...data, missing]);
}
