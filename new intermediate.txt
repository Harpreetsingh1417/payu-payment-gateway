from google.cloud import spanner


# Google Cloud project
PROJECT_ID = "invertible-now-413616"

# Spanner details
INSTANCE_ID = "your-spanner-instance"
DATABASE_ID = "operational-data"


# Create Spanner client
spanner_client = spanner.Client(project=PROJECT_ID)

# Get Spanner instance
instance = spanner_client.instance(INSTANCE_ID)

# Get database
database = instance.database(DATABASE_ID)


# DDL schemas for Products and Variants
ddl_statements = [

    """
    CREATE TABLE Products (
        ProductId STRING(36) NOT NULL,
        ProductName STRING(200) NOT NULL,
        Description STRING(MAX),
        CreatedAt TIMESTAMP NOT NULL
    ) PRIMARY KEY (ProductId)
    """,

    """
    CREATE TABLE Variants (
        VariantId STRING(36) NOT NULL,
        ProductId STRING(36) NOT NULL,
        VariantName STRING(200) NOT NULL,
        Price NUMERIC NOT NULL,
        CreatedAt TIMESTAMP NOT NULL
    ) PRIMARY KEY (VariantId),
      CONSTRAINT FK_Variants_Products
      FOREIGN KEY (ProductId)
      REFERENCES Products (ProductId)
    """
]


print("Creating Products and Variants tables...")

# Apply the DDL
operation = database.update_ddl(ddl_statements)

# Wait until the operation is complete
operation.result()

print("Tables created successfully!")
